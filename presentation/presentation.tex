\begin{frame}{Motivation & Goals}
    \begin{itemize}
        \item \textbf{Problem:} AI hardware power consumption is skyrocketing.
        \item \textbf{Solution:} Passive Optical Computing (Speed of light, Zero energy).
        \item \textbf{Challenge:} "Curriculum Mismatch" causes training instability.
        \item \textbf{Result:} A \textbf{Hybrid Architecture} that beats the CNN baseline.
    \end{itemize}
\end{frame}
